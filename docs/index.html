Consumo

# Tabla de equipos
curl -s https://raw.githubusercontent.com/Norwyx/fpc-api/main/docs/v1/teams.json

# Una temporada completa (posiciones, partidos, goleadores, asistencias)
curl -s https://raw.githubusercontent.com/Norwyx/fpc-api/main/docs/v1/seasons/2026-i.json

# Con GitHub Pages activado (Settings → Pages → GitHub Actions)
curl -s https://Norwyx.github.io/fpc-api/v1/index.json

# JavaScript (CORS habilitado)
const data = await fetch('https://Norwyx.github.io/fpc-api/v1/teams.json').then(r => r.json());
